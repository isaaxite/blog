您访问的页面迷路了，即将为您跳转回[主页](/blog/)

或者，试试搜索这些相近的文章？

<ul id="suggestions" class="suggestion-list">
  <li>正在寻找相似文章...</li>
</ul>

<script>
(function() {
  // 获取当前错误的路径
  var path = window.location.pathname;
  console.log('当前错误路径:', path);

  // 获取当前页面显示的域名，用于构建完整链接
  var baseUrl = window.location.origin;

  // 1. 设置定时跳转（可选，比如5秒后跳转到主页）
  // setTimeout(function() {
  //   window.location.href = '/blog/'; // 修改为你主页的路径
  // }, 15000); // 15000毫秒 = 15秒

  // 2. 尝试从 sitemap 获取推荐文章
  var suggestionList = document.getElementById('suggestions');

  // 获取 sitemap 文件
  fetch('/blog/sitemap.xml') // 确认你的 sitemap 路径是否正确
    .then(response => response.text())
    .then(str => new window.DOMParser().parseFromString(str, "text/xml"))
    .then(data => {
      var urls = data.querySelectorAll('url loc');
      var suggestions = [];
      var pathSegments = path.split('/').filter(s => s); // 获取路径的关键词

      // 简单的匹配规则：遍历所有链接，找到路径中包含当前路径关键词的链接
      urls.forEach(function(locNode) {
        var url = locNode.textContent;
        // 跳过自身（404页面）和主页
        if (url.includes('/404.html') || url === baseUrl + '/blog/') return; 

        // 关键词匹配逻辑：如果当前错误路径的任一段落出现在文章链接中，就视为相关
        pathSegments.forEach(segment => {
          if (segment.length > 3 && url.includes(segment) && !suggestions.includes(url)) { // 避免过短关键词和重复
              suggestions.push(url);
          }
        });
      });

      // 显示推荐结果
      if (suggestions.length > 0) {
        suggestionList.innerHTML = ''; // 清空 "正在寻找..."
        suggestions.slice(0, 5).forEach(function(url) { // 最多显示5条
          var li = document.createElement('li');
          var a = document.createElement('a');
          a.href = url;
          // 从链接中提取一个友好的标题（例如取最后一部分）
          var titlePart = url.split('/').filter(s => s).pop().replace(/-/g, ' ') || '相关文章';
          a.textContent = decodeURIComponent(titlePart);
          li.appendChild(a);
          suggestionList.appendChild(li);
        });
      } else {
        suggestionList.innerHTML = '<li>没有找到非常相似的文章，<a href="/blog/">返回主页</a>看看吧。</li>';
      }
  })
  .catch(error => {
    console.error('加载 sitemap 失败:', error);
    suggestionList.innerHTML = '<li>无法加载文章推荐，<a href="/blog/">点击这里返回主页</a>。</li>';
  });
})();
</script>
